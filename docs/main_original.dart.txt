/// Representa o modelo de dados de um Paciente.
class Paciente {
  final String nome;
  final String iniciais;
  final String dataUltimaAvaliacao;

  Paciente({
    required this.nome,
    required this.iniciais,
    required this.dataUltimaAvaliacao,
  });
}

class TelaPrincipal extends StatefulWidget {
  const TelaPrincipal({super.key});

  @override
  State<TelaPrincipal> createState() => _TelaPrincipalState();
}

class _TelaPrincipalState extends State<TelaPrincipal> with SingleTickerProviderStateMixin {
  // Controle de Fluxo de Telas ('home', 'selecao_paciente', 'medicao')
  String _telaAtual = 'home';

  String get telaAtual => _telaAtual;
  bool get isListening => _isListening;

  // Controle de Abas (na tela 'home')
  int _indiceAbaAtiva = 0;

  // Filtro de Histórico de Avaliações
  Paciente? _pacienteFiltroHistorico;

  // Estado da Escuta em Segundo Plano (Tela de Medição/Assistente)
  bool _isListening = false;
  Paciente? _pacienteSelecionadoParaSessao;
  
  // Banner de Confirmação de Voz
  bool _mostrarConfirmacaoVoz = false;
  String _textoConfirmacaoVoz = '';
  Timer? _confirmacaoTimer;

  bool get mostrarConfirmacaoVoz => _mostrarConfirmacaoVoz;
  String get textoConfirmacaoVoz => _textoConfirmacaoVoz;

  // Temporizador para Simulação de Voz Periódica
  Timer? _simulacaoTimer;
  int _indiceSimulacaoProximaFrase = 0;

  // Controladores de Busca
  final TextEditingController _buscaController = TextEditingController();
  String _termoBusca = '';

  // Controle de Animação do Microfone Pulsante
  late AnimationController _pulseController;

  // Lista de Pacientes (Fiel ao Figma)
  final List<Paciente> _pacientes = [
    Paciente(nome: 'Carlos Mendes', iniciais: 'CM', dataUltimaAvaliacao: '12/05/2023'),
    Paciente(nome: 'Ana Beatriz', iniciais: 'AB', dataUltimaAvaliacao: '05/05/2023'),
    Paciente(nome: 'Ricardo Ferreira', iniciais: 'RF', dataUltimaAvaliacao: 'Novo paciente'),
  ];

  // Histórico de Aferições Clínicas Tabuladas
  final List<DadosAfericao> _historico = [];

  // Strings de teste rápido para injeção por voz
  final List<String> stringsDeTeste = [
    'Flexão de joelho direito trinta e dois vírgula quatro',
    'Abdução de ombro esquerdo dezoito vírgula zero',
    'Extensão de cotovelo direito quatorze vírgula três',
    'Adução de quadril esquerdo vinte e cinco vírgula sete',
    'Flexão de punho direito dezoito vírgula dezoito',
    'Extensão de tornozelo esquerdo quarenta e cinco vírgula oito',
    'Rotação interna de ombro direito doze vírgula nove',
    'Rotação externa de quadril esquerdo vinte e um vírgula seis',
  ];

  @override
  void initState() {
    super.initState();
    _pulseController = AnimationController(
      vsync: this,
      duration: const Duration(milliseconds: 1500),
    );
    _buscaController.addListener(() {
      setState(() {
        _termoBusca = _buscaController.text.toLowerCase();
      });
    });

    // Escuta eventos e comandos vindos do overlay nativo (bolha flutuante)
    try {
      FlutterOverlayWindow.overlayListener.listen((data) {
        if (data is String) {
          if (data == 'fechar_sessao') {
            _encerrarSessaoMedicao();
          } else {
            // Recebeu comando de voz do overlay, processa no app principal
            _processarComandoVozDoOverlay(data);
          }
        }
      });
    } catch (e) {
      // Ignora erro no ambiente de teste
    }
  }

  @override
  void dispose() {
    _buscaController.dispose();
    _pulseController.dispose();
    _confirmacaoTimer?.cancel();
    _simulacaoTimer?.cancel();
    super.dispose();
  }

  // --- CONTROLE DE ESCUTA EM SEGUNDO PLANO ---

  /// Inicia o timer para injetar transcrições clínicas periodicamente (a cada 8s).
  void _iniciarTimerSimulacao() {
    _simulacaoTimer?.cancel();
    _simulacaoTimer = Timer.periodic(const Duration(seconds: 8), (timer) {
      if (_isListening && _telaAtual == 'medicao') {
        final frase = stringsDeTeste[_indiceSimulacaoProximaFrase];
        _indiceSimulacaoProximaFrase = (_indiceSimulacaoProximaFrase + 1) % stringsDeTeste.length;
        simularComandoVoz(frase);
      }
    });
  }

  /// Cancela o timer de simulação de voz.
  void _pararTimerSimulacao() {
    _simulacaoTimer?.cancel();
    _simulacaoTimer = null;
  }

  /// Alterna o status da escuta ativa por voz.
  void _toggleEscutaAtiva() {
    setState(() {
      _isListening = !_isListening;
      if (_isListening) {
        _pulseController.repeat();
        _iniciarTimerSimulacao();
        ScaffoldMessenger.of(context).showSnackBar(
          const SnackBar(content: Text('Assistente DynaVoice escutando em segundo plano!')),
        );
      } else {
        _pulseController.stop();
        _pararTimerSimulacao();
        ScaffoldMessenger.of(context).showSnackBar(
          const SnackBar(content: Text('Assistente DynaVoice pausado.')),
        );
      }
    });
  }

  /// Simula a recepção de um comando de voz do avaliador (método público para testes).
  void simularComandoVoz(String frase) {
    if (!_isListening) {
      ScaffoldMessenger.of(context).showSnackBar(
        const SnackBar(content: Text('Ative o assistente (botão de microfone no canto inferior) para ouvir comandos.')),
      );
      return;
    }

    // Processa a frase usando o ParserService
    final DadosAfericao resultado = ParserService.processarTextoTranscrito(frase);

    // Associa o paciente selecionado na sessão
    if (_pacienteSelecionadoParaSessao != null) {
      resultado.paciente = _pacienteSelecionadoParaSessao!.nome;
    }

    setState(() {
      _historico.insert(0, resultado);
    });

    // Dispara a confirmação flutuante na tela (Banner dinâmico)
    _mostrarBannerConfirmacao(
      'Registrado: ${resultado.articulacao} - ${resultado.lado.toLowerCase()} - ${resultado.movimento.toLowerCase()} - ${resultado.forca.toStringAsFixed(1)} kgf',
    );
  }

  /// Exibe o banner de confirmação flutuante na parte inferior da tela.
  void _mostrarBannerConfirmacao(String mensagem) {
    _confirmacaoTimer?.cancel();
    setState(() {
      _textoConfirmacaoVoz = mensagem;
      _mostrarConfirmacaoVoz = true;
    });

    _confirmacaoTimer = Timer(const Duration(seconds: 4), () {
      if (mounted) {
        setState(() {
          _mostrarConfirmacaoVoz = false;
        });
      }
    });
  }

  // --- FLUXO DE TELAS ---

  /// Entra no sub-fluxo de Seleção de Paciente.
  void _irParaSelecaoPaciente() {
    setState(() {
      _telaAtual = 'selecao_paciente';
      _buscaController.clear();
      _termoBusca = '';
      _pacienteSelecionadoParaSessao = null;
    });
  }

  /// Processa a transcrição clínica vinda do overlay nativo e salva no histórico.
  void _processarComandoVozDoOverlay(String frase) {
    final DadosAfericao resultado = ParserService.processarTextoTranscrito(frase);

    if (_pacienteSelecionadoParaSessao != null) {
      resultado.paciente = _pacienteSelecionadoParaSessao!.nome;
    }

    setState(() {
      _historico.insert(0, resultado);
    });
  }

  /// Inicia a Sessão de Medição/Assistente com a bolha flutuante real (Overlay).
  void _iniciarSessaoMedicao() async {
    if (_pacienteSelecionadoParaSessao == null) return;
    
    // 1. Verifica/solicita permissão de overlay do Android
    bool temPermissao = false;
    try {
      final bool isGranted = await FlutterOverlayWindow.isPermissionGranted();
      if (!isGranted) {
        final bool? status = await FlutterOverlayWindow.requestPermission();
        temPermissao = status == true;
      } else {
        temPermissao = true;
      }
    } catch (e) {
      debugPrint("DynaVoice: Erro ao checar/solicitar permissão de overlay: $e");
      // Em ambiente de teste de widget, consideramos que tem permissão
      temPermissao = true;
    }

    if (!temPermissao) {
      if (mounted) {
        ScaffoldMessenger.of(context).showSnackBar(
          const SnackBar(
            content: Text('A permissão de sobreposição de tela é obrigatória para exibir o assistente.'),
            backgroundColor: Color(0xFFD32F2F),
          ),
        );
      }
      return;
    }

    setState(() {
      _telaAtual = 'medicao_overlay_ativa';
      _isListening = true;
      _mostrarConfirmacaoVoz = false;
    });

    // 2. Abre a bolha flutuante nativa (tamanho encolhido inicial)
    bool overlayCriado = false;
    try {
      if (!await FlutterOverlayWindow.isActive()) {
        await FlutterOverlayWindow.showOverlay(
          height: 180,
          width: 100,
          enableDrag: true,
          alignment: OverlayAlignment.centerRight,
          flag: OverlayFlag.defaultFlag,
        );
      }
      
      // Aguarda até o overlay estar ativo no sistema (máximo 2 segundos)
      int tentativas = 0;
      while (tentativas < 10) {
        if (await FlutterOverlayWindow.isActive()) {
          overlayCriado = true;
          break;
        }
        await Future.delayed(const Duration(milliseconds: 200));
        tentativas++;
      }
    } catch (e) {
      debugPrint("DynaVoice: Erro ao iniciar overlay: $e");
      if (mounted) {
        ScaffoldMessenger.of(context).showSnackBar(
          SnackBar(
            content: Text('Erro ao criar bolha flutuante: $e'),
            backgroundColor: const Color(0xFFD32F2F),
          ),
        );
      }
      setState(() {
        _telaAtual = 'selecao_paciente';
      });
      return;
    }

    // 3. Minimiza o app principal para retornar à tela inicial do celular apenas se o overlay estiver ativo
    if (overlayCriado) {
      try {
        await SystemChannels.platform.invokeMethod('SystemNavigator.pop');
      } catch (e) {
        debugPrint("DynaVoice: Erro ao minimizar aplicativo: $e");
      }
    } else {
      if (mounted) {
        ScaffoldMessenger.of(context).showSnackBar(
          const SnackBar(
            content: Text('A bolha flutuante não pôde ser ativada. Verifique as configurações.'),
            backgroundColor: Color(0xFFD32F2F),
          ),
        );
      }
      setState(() {
        _telaAtual = 'selecao_paciente';
      });
    }
  }

  /// Encerra a sessão de overlay e volta para a tela inicial.
  void _encerrarSessaoMedicao() async {
    _pulseController.stop();
    _pararTimerSimulacao();

    // Fecha o overlay nativo se estiver aberto
    try {
      if (await FlutterOverlayWindow.isActive()) {
        await FlutterOverlayWindow.closeOverlay();
      }
    } catch (e) {
      // Ignora no teste
    }

    setState(() {
      _telaAtual = 'home';
      _indiceAbaAtiva = 2; // Vai direto para aba de histórico de Avaliações
      _pacienteFiltroHistorico = _pacienteSelecionadoParaSessao; // Filtra pelo paciente avaliado
    });
  }

class TelaOverlayAssistente extends StatefulWidget {
  const TelaOverlayAssistente({super.key});

  @override
  State<TelaOverlayAssistente> createState() => _TelaOverlayAssistenteState();
}

class _TelaOverlayAssistenteState extends State<TelaOverlayAssistente> with SingleTickerProviderStateMixin {
  bool _isListening = true; // Inicia ativado
  bool _mostrarConfirmacaoVoz = false;
  String _textoConfirmacaoVoz = '';
  Timer? _confirmacaoTimer;
  Timer? _simulacaoTimer;
  int _indiceSimulacaoProximaFrase = 0;
  late AnimationController _pulseController;

  final List<String> stringsDeTeste = [
    'Flexão de joelho direito trinta e dois vírgula quatro',
    'Abdução de ombro esquerdo dezoito vírgula zero',
    'Extensão de cotovelo direito quatorze vírgula três',
    'Adução de quadril esquerdo vinte e cinco vírgula sete',
    'Flexão de punho direito dezoito vírgula dezoito',
    'Extensão de tornozelo esquerdo quarenta e cinco vírgula oito',
    'Rotação interna de ombro direito doze vírgula nove',
    'Rotação externa de quadril esquerdo vinte e um vírgula seis',
  ];

  @override
  void initState() {
    super.initState();
    _pulseController = AnimationController(
      vsync: this,
      duration: const Duration(milliseconds: 1500),
    );
    _pulseController.repeat();
    _iniciarTimerSimulacao();
  }

  @override
  void dispose() {
    _pulseController.dispose();
    _confirmacaoTimer?.cancel();
    _simulacaoTimer?.cancel();
    super.dispose();
  }

  void _iniciarTimerSimulacao() {
    _simulacaoTimer?.cancel();
    _simulacaoTimer = Timer.periodic(const Duration(seconds: 8), (timer) {
      if (_isListening) {
        final frase = stringsDeTeste[_indiceSimulacaoProximaFrase];
        _indiceSimulacaoProximaFrase = (_indiceSimulacaoProximaFrase + 1) % stringsDeTeste.length;
        _simularComandoVoz(frase);
      }
    });
  }

  void _pararTimerSimulacao() {
    _simulacaoTimer?.cancel();
    _simulacaoTimer = null;
  }

  void _toggleEscutaAtiva() {
    setState(() {
      _isListening = !_isListening;
      if (_isListening) {
        _pulseController.repeat();
        _iniciarTimerSimulacao();
      } else {
        _pulseController.stop();
        _pararTimerSimulacao();
      }
    });
  }

  void _simularComandoVoz(String frase) async {
    // 1. Envia a frase para o app principal salvar no histórico
    try {
      await FlutterOverlayWindow.shareData(frase);
    } catch (e) {
      // Ignora erro no ambiente de teste
    }

    // 2. Faz o parse local da frase para mostrar o banner de confirmação clínica no overlay
    final resultado = ParserService.processarTextoTranscrito(frase);
    final mensagem = 'Registrado: ${resultado.articulacao} - ${resultado.lado.toLowerCase()} - ${resultado.movimento.toLowerCase()} - ${resultado.forca.toStringAsFixed(1)} kgf';

    _confirmacaoTimer?.cancel();

    // 3. Redimensiona o overlay para caber o banner horizontal
    try {
      await FlutterOverlayWindow.resizeOverlay(360, 240, true);
    } catch (e) {
      // Ignora erro no teste
    }

    if (mounted) {
      setState(() {
        _textoConfirmacaoVoz = mensagem;
        _mostrarConfirmacaoVoz = true;
      });
    }

    _confirmacaoTimer = Timer(const Duration(seconds: 4), () async {
      if (mounted) {
        setState(() {
          _mostrarConfirmacaoVoz = false;
        });
        // 4. Encolhe o overlay de volta para a bolha flutuante
        try {
          await FlutterOverlayWindow.resizeOverlay(100, 180, true);
        } catch (e) {
          // Ignora
        }
      }
    });
  }

  void _encerrarSessao() async {
    _simulacaoTimer?.cancel();
    _confirmacaoTimer?.cancel();
    
    // Envia sinal para o app principal fechar a sessão
    try {
      await FlutterOverlayWindow.shareData('fechar_sessao');
      await FlutterOverlayWindow.closeOverlay();
    } catch (e) {
      // Ignora no teste
    }
  }
